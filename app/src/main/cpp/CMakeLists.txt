cmake_minimum_required(VERSION 3.22)
project(daylight_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Keep the sideloadable debug APK fast enough for real on-device inference.
set(CMAKE_C_FLAGS_DEBUG "-g -O2" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-g -O2" CACHE STRING "" FORCE)
set_property(GLOBAL PROPERTY JOB_POOLS native_compile=2)
set(CMAKE_JOB_POOL_COMPILE native_compile)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run tools/prepare-native.ps1 first (pinned llama.cpp source).")
endif()
execute_process(COMMAND git -C "${LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT LLAMA_REV STREQUAL "889f2ec96deacbc2d8a9a67dbe181eaa43a2fb3b")
    message(FATAL_ERROR "Unexpected llama.cpp revision. Run tools/prepare-native.ps1.")
endif()
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(daylight_inference SHARED inference.cpp)
target_link_libraries(daylight_inference PRIVATE llama android log)
target_link_options(daylight_inference PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")

# BSD-licensed, offline RAR/CBR normalization. Only the RAR readers are enabled at runtime.
set(ARCHIVE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native/libarchive")
if(NOT EXISTS "${ARCHIVE_SOURCE}/libarchive/archive.h")
    message(FATAL_ERROR "Run tools/prepare-native.ps1 first (pinned libarchive source).")
endif()
execute_process(COMMAND git -C "${ARCHIVE_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE ARCHIVE_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT ARCHIVE_REV STREQUAL "0c48412be2d94afe1859a807ea7896a18d8900bd")
    message(FATAL_ERROR "Unexpected libarchive revision. Run tools/prepare-native.ps1.")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
foreach(feature WERROR OPENSSL NETTLE MBEDTLS MD LIBB2 LZ4 LZO LZMA ZSTD BZip2 LIBXML2 BSDXML EXPAT WIN32_XMLLITE PCREPOSIX PCRE2POSIX LIBGCC CNG TAR CPIO CAT UNZIP XATTR ACL ICONV TEST INSTALL)
    set(ENABLE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_ZLIB ON CACHE BOOL "" FORCE)
add_subdirectory("${ARCHIVE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/archive" EXCLUDE_FROM_ALL)
add_library(daylight_archive SHARED archive.cpp)
target_include_directories(daylight_archive PRIVATE "${ARCHIVE_SOURCE}/libarchive" "${CMAKE_CURRENT_BINARY_DIR}/archive")
target_link_libraries(daylight_archive PRIVATE archive_static log)
target_link_options(daylight_archive PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
